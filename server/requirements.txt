# InfinityServer runtime deps.
#
# SQLite (the default backend, INFINITY_DB=sqlite) needs nothing here — it's stdlib.
# Postgres (INFINITY_DB=postgres, the hosted path) needs psycopg 3. The [binary] extra
# pulls a prebuilt libpq wheel (incl. aarch64, matching the OCI ARM VM) so there's no
# system libpq/build toolchain requirement.
psycopg[binary]>=3.1

# MariaDB (INFINITY_DB=mysql) is the SHARED database: SwordShine's schema, with our tables
# namespaced and our accounts/characters/char_items mapped onto its users/users_items. That
# path needs a driver, and bcrypt because the shared `Hash` column is read by SwordShine's
# PHP login (password_verify), which understands bcrypt and nothing else.
PyMySQL>=1.1
bcrypt>=4.0
